[
  {
    $match: {
      tags: {
        $in: [
          "Couple",
          "Family with young children",
          "Family with older children",
          "Solo traveler",
          "Business trip"
        ]
      }
    }
  },
  {
    $lookup: {
      from: "v1_hotels",
      localField: "hotel_id",
      foreignField: "_id",
      as: "hotel"
    }
  },
  {
    $unwind: "$hotel"
  },
  {
    $set: {
      guest_types: {
        $setUnion: [
          {
            $cond: [
              { $in: ["Couple", "$tags"] },
              ["couple"],
              []
            ]
          },
          {
            $cond: [
              {
                $or: [
                  { $in: ["Family with young children", "$tags"] },
                  { $in: ["Family with older children", "$tags"] }
                ]
              },
              ["family"],
              []
            ]
          },
          {
            $cond: [
              { $in: ["Solo traveler", "$tags"] },
              ["solo"],
              []
            ]
          },
          {
            $cond: [
              { $in: ["Business trip", "$tags"] },
              ["business"],
              []
            ]
          }
        ]
      }
    }
  },
  {
    $match: {
      "hotel.city": { $ne: null },
      $expr: {
        $gt: [
          { $size: "$guest_types" },
          0
        ]
      }
    }
  },
  {
    $project: {
      _id: 1,
      review_id: "$_id",
      hotel: {
        id: "$hotel._id",
        name: "$hotel.name",
        address: "$hotel.address",
        city: "$hotel.city",
        country: "$hotel.country"
      },
      reviewer_score: 1,
      tags: 1,
      guest_types: 1
    }
  },
  {
    $merge: {
      into: "v2_reviews_guest_type",
      whenMatched: "replace",
      whenNotMatched: "insert"
    }
  }
]
