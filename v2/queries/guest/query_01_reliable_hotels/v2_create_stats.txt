[
  {
    $lookup: {
      from: "v1_hotels",
      localField: "hotel_id",
      foreignField: "_id",
      as: "hotel"
    }
  },
  {
    $unwind: "$hotel"
  },
  {
    $match: {
      "hotel.city": { $ne: null }
    }
  },
  {
    $group: {
      _id: {
        city: "$hotel.city",
        hotel_id: "$hotel._id"
      },
      hotel_name: { $first: "$hotel.name" },
      address: { $first: "$hotel.address" },
      city: { $first: "$hotel.city" },
      country: { $first: "$hotel.country" },
      score_sum: { $sum: "$reviewer_score" },
      average_reviewer_score: { $avg: "$reviewer_score" },
      review_count: { $sum: 1 }
    }
  },
  {
    $group: {
      _id: "$city",
      city_score_sum: { $sum: "$score_sum" },
      city_review_count: { $sum: "$review_count" },
      hotels: {
        $push: {
          hotel_id: "$_id.hotel_id",
          hotel_name: "$hotel_name",
          address: "$address",
          city: "$city",
          country: "$country",
          average_reviewer_score: "$average_reviewer_score",
          review_count: "$review_count"
        }
      }
    }
  },
  {
    $set: {
      city_average_score: {
        $divide: ["$city_score_sum", "$city_review_count"]
      }
    }
  },
  {
    $unwind: "$hotels"
  },
  {
    $project: {
      _id: "$hotels.hotel_id",
      hotel_id: "$hotels.hotel_id",
      hotel_name: "$hotels.hotel_name",
      address: "$hotels.address",
      city: "$hotels.city",
      country: "$hotels.country",
      average_reviewer_score: {
        $round: ["$hotels.average_reviewer_score", 2]
      },
      review_count: "$hotels.review_count",
      city_average_score: {
        $round: ["$city_average_score", 2]
      },
      minimum_reviews: 100,
      reliability_score: {
        $round: [
          {
            $add: [
              {
                $multiply: [
                  {
                    $divide: [
                      "$hotels.review_count",
                      { $add: ["$hotels.review_count", 100] }
                    ]
                  },
                  "$hotels.average_reviewer_score"
                ]
              },
              {
                $multiply: [
                  {
                    $divide: [
                      100,
                      { $add: ["$hotels.review_count", 100] }
                    ]
                  },
                  "$city_average_score"
                ]
              }
            ]
          },
          4
        ]
      },
      calculated_at: "$$NOW"
    }
  },
  {
    $merge: {
      into: "v2_hotel_reliability_stats",
      whenMatched: "replace",
      whenNotMatched: "insert"
    }
  }
]
