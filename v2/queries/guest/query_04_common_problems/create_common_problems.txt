db.v1_reviews.aggregate(
  [
    {
      $lookup: {
        from: "v1_hotels",
        localField: "hotel_id",
        foreignField: "_id",
        as: "hotel"
      }
    },
    {
      $unwind: "$hotel"
    },
    {
      $match: {
        negative_review: {
          $ne: "No Negative"
        }
      }
    },
    {
      $addFields: {
        negative_review_lower: {
          $toLower: "$negative_review"
        }
      }
    },
    {
      $addFields: {
        negative_problem_attributes: {
          $setUnion: [
            {
              $cond: [
                {
                  $regexMatch: {
                    input: "$negative_review_lower",
                    regex: "noise|noisy|loud|soundproof"
                  }
                },
                [
                  {
                    problem_type: "noise"
                  }
                ],
                []
              ]
            },
            {
              $cond: [
                {
                  $regexMatch: {
                    input: "$negative_review_lower",
                    regex: "small room|tiny room|room was small|rooms are small"
                  }
                },
                [
                  {
                    problem_type: "small_room"
                  }
                ],
                []
              ]
            },
            {
              $cond: [
                {
                  $regexMatch: {
                    input: "$negative_review_lower",
                    regex: "dirty|cleanliness|unclean|not clean|dust|smell"
                  }
                },
                [
                  {
                    problem_type: "cleanliness"
                  }
                ],
                []
              ]
            },
            {
              $cond: [
                {
                  $regexMatch: {
                    input: "$negative_review_lower",
                    regex: "bed|mattress|pillow"
                  }
                },
                [
                  {
                    problem_type: "bed"
                  }
                ],
                []
              ]
            },
            {
              $cond: [
                {
                  $regexMatch: {
                    input: "$negative_review_lower",
                    regex: "air conditioning|aircondition|air con|a/c| ac |heating|climate"
                  }
                },
                [
                  {
                    problem_type: "air_conditioning"
                  }
                ],
                []
              ]
            },
            {
              $cond: [
                {
                  $regexMatch: {
                    input: "$negative_review_lower",
                    regex: "wifi|wi fi|wi-fi|internet"
                  }
                },
                [
                  {
                    problem_type: "wifi"
                  }
                ],
                []
              ]
            }
          ]
        }
      }
    },
    {
      $match: {
        $expr: {
          $gt: [
            {
              $size: "$negative_problem_attributes"
            },
            0
          ]
        }
      }
    },
    {
      $project: {
        _id: 0,
        review_id: "$_id",
        review_date: 1,
        reviewer_score: 1,
        negative_review: 1,

        hotel: {
          hotel_id: "$hotel._id",
          name: "$hotel.name",
          address: "$hotel.address",
          city: "$hotel.city",
          country: "$hotel.country",
          total_hotel_reviews: "$hotel.total_number_of_reviews"
        },

        negative_problem_attributes: 1,

        schema_pattern: "attribute",
        attribute_description: "Detected negative review problems are stored as an array of problem attributes."
      }
    },
    {
      $merge: {
        into: "v2_reviews_common_problems",
        whenMatched: "replace",
        whenNotMatched: "insert"
      }
    }
  ],
  {
    allowDiskUse: true
  }
);
